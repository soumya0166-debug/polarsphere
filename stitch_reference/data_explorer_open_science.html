Python SDK Query Example xarray / ncpor-py 
import ncpor
import xarray as xr

# Initialize client for Open Polar Archive
client = ncpor.Client(auth="ANON_OPEN")

# Fetch Larsemann GNSS 2024-2025 time slice
dataset = client.datasets.get("10.5065/NCPOR.ANT.2025.09")
ds = xr.open_dataset(dataset.opendap_url)

# Compute seasonal vertical displacement
uplift_rate = ds['dh_dt'].sel(station='BHARATI_GPS_01').mean()
print(f"Current Crustal Rebound: {uplift_rate.values:.2f} mm/yr")
